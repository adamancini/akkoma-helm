   kubectl port-forward svc/{{ include "akkoma.fullname" . }} 4000:4000

   kubectl exec -it deployment/{{ include "akkoma.fullname" . }} -- \
     /opt/akkoma/bin/pleroma_ctl user new admin {{ .Values.akkoma.adminEmail }} --admin
